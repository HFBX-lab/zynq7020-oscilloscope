cmake_minimum_required(VERSION 3.20)
project(SP2BoardLVGL C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(path LVGL_ROOT UI_ROOT LEGACY_ROOT BSP_INCLUDE BSP_LIB)
  file(TO_CMAKE_PATH "${${path}}" ${path})
endforeach()
add_compile_options(-pipe -O2 -g -mcpu=cortex-a9 -mfpu=vfpv3 -mfloat-abi=hard -ffunction-sections -fdata-sections)
file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${LVGL_ROOT}/src/*.c")
add_library(lvgl STATIC ${LVGL_SOURCES})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
if(SCOPE_OS STREQUAL "freertos")
  target_compile_definitions(lvgl PUBLIC SCOPE_FREERTOS=1)
endif()
target_include_directories(lvgl PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${LVGL_ROOT}" "${BSP_INCLUDE}")
add_executable(scope_board app.c board.c portable.c
  "${UI_ROOT}/scope_calibration.c" "${UI_ROOT}/scope_calibration_profile.c"
  "${UI_ROOT}/scope_ui.c" "${UI_ROOT}/scope_model.c" "${UI_ROOT}/scope_trigger.c" "${UI_ROOT}/scope_cursor.c" "${UI_ROOT}/scope_acquisition.c" "${UI_ROOT}/scope_measurement.c" "${UI_ROOT}/scope_autoset.c" "${UI_ROOT}/scope_record.c" "${UI_ROOT}/scope_settings.c" "${UI_ROOT}/scope_spectrum.c"
  "${UI_ROOT}/fonts/scope_font_14.c" "${UI_ROOT}/fonts/scope_font_16.c" "${UI_ROOT}/fonts/scope_font_24.c"
  "${LEGACY_ROOT}/frame_source.c" "${LEGACY_ROOT}/memory_probe.c")
if(SCOPE_TOUCH_ROOT)
  target_sources(scope_board PRIVATE "${SCOPE_TOUCH_ROOT}/touch.c" "${SCOPE_TOUCH_ROOT}/i2c_bitbang.c" "${SCOPE_TOUCH_ROOT}/board_touch.c")
  target_include_directories(scope_board PRIVATE "${SCOPE_TOUCH_ROOT}")
  target_compile_definitions(scope_board PRIVATE SCOPE_PHYSICAL_TOUCH=1)
endif()
target_include_directories(scope_board PRIVATE "${UI_ROOT}" "${LEGACY_ROOT}")
if(SCOPE_ENGINEERING_CALIBRATION)
  target_compile_definitions(scope_board PRIVATE SCOPE_ENGINEERING_CALIBRATION=1)
endif()
if(SCOPE_ADC_ROOT)
  target_sources(scope_board PRIVATE "${SCOPE_ADC_ROOT}/board_adc_live.c" "${SCOPE_ADC_ROOT}/record_stats.c")
  target_include_directories(scope_board PRIVATE "${SCOPE_ADC_ROOT}")
  target_compile_definitions(scope_board PRIVATE SCOPE_PHYSICAL_ADC=1)
endif()
if(SCOPE_STORAGE_ROOT)
  file(TO_CMAKE_PATH "${SCOPE_STORAGE_ROOT}" SCOPE_STORAGE_ROOT)
  file(TO_CMAKE_PATH "${SCOPE_STORAGE_SDPS_LIB}" SCOPE_STORAGE_SDPS_LIB)
  target_sources(scope_board PRIVATE "${UI_ROOT}/scope_storage.c" "${UI_ROOT}/scope_storage_ui.c" "${SCOPE_STORAGE_ROOT}/board_storage.c")
  target_include_directories(scope_board PRIVATE "${SCOPE_STORAGE_ROOT}")
  target_compile_definitions(scope_board PRIVATE SCOPE_TF_STORAGE=1)
endif()
target_compile_options(scope_board PRIVATE -Wall -Wextra)
target_link_directories(scope_board PRIVATE "${BSP_LIB}")
target_link_options(scope_board PRIVATE -mcpu=cortex-a9 -mfpu=vfpv3 -mfloat-abi=hard
  "-specs=${CMAKE_CURRENT_SOURCE_DIR}/Xilinx.spec" "-Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/lscript.ld"
  "-Wl,-Map,${CMAKE_BINARY_DIR}/scope_lvgl_${SCOPE_OS}.map" -Wl,--gc-sections)
target_link_libraries(scope_board PRIVATE lvgl -Wl,--start-group)
if(SCOPE_STORAGE_ROOT)
  target_link_libraries(scope_board PRIVATE "${SCOPE_STORAGE_SDPS_LIB}" xilffs)
endif()
target_link_libraries(scope_board PRIVATE xil)
if(SCOPE_OS STREQUAL "freertos")
  target_link_libraries(scope_board PRIVATE freertos)
endif()
target_link_libraries(scope_board PRIVATE gcc c m -Wl,--end-group)
set_target_properties(scope_board PROPERTIES OUTPUT_NAME "scope_lvgl_${SCOPE_OS}" SUFFIX ".elf")
